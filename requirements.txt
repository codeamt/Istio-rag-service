from fastapi import FastAPI, BackgroundTasks
import httpx
import hashlib 
from datetime import datetime
from pydantic import BaseModel

app = FastAPI()

class ScrapeRequest(BaseModel):
    query: str 
    platforms: list = ["twitter", "bluesky", "threads"]
    max_posts: int = 100 

@app.post("/scrape")
async def scrape_and_index(request: ScrapeRequest, background_tasks: BackgroundTasks):
    """
    Endpoint to scrape social media platforms and index the data.
    """
    background_tasks.add_task(run_scrapers, request) 
    return {"status": "Scraping started", "query": request.query, "platforms": request.platforms, "max_posts": request.max_posts}

async def run_scrapers(request: ScrapeRequest):
    """
    Function to run the scrapers for the specified platforms.
    """
    posts = []
    for platform in request.platforms:
        if platform == "twitter":
            posts.extend(await scrape_twitter(request.query, request.max_posts))
        elif platform == "bluesky":
            posts.extend(await scrape_bluesky(request.query, request.max_posts))
        elif platform == "threads":
            posts.extend(await scrape_threads(request.query, request.max_posts))

    #Index in Qdrant 
    await index_posts(posts)

async def scrape_twitter(query: str, max_posts: int):
    """
    Function to scrape Twitter for the given query.
    """
    # Placeholder for Twitter scraping logic
    return [{
        "id": hashlib.sha256(f"twitter_{query}_{i}".encode()).hexdigest(),
        "text": f"Tweet {i} about {query}",
        "platform": "twitter",
        "timestamp": datetime.now().isoformat()
    } for i in range(max_posts)]

async def scrape_bluesky(query: str, max_posts: int):
    """
    Function to scrape Bluesky for the given query.
    """
    # Placeholder for Bluesky scraping logic
    return [{
        "id": hashlib.sha256(f"bluesky_{query}_{i}".encode()).hexdigest(),
        "text": f"Bluesky post {i} about {query}",
        "platform": "bluesky",
        "timestamp": datetime.now().isoformat()
    } for i in range(max_posts)]

async def scrape_threads(query: str, max_posts: int):
    """
    Function to scrape Threads for the given query.
    """
    # Placeholder for Threads scraping logic
    return [{
        "id": hashlib.sha256(f"threads_{query}_{i}".encode()).hexdigest(),
        "text": f"Threads post {i} about {query}",
        "platform": "threads",
        "timestamp": datetime.now().isoformat()
    } for i in range(max_posts)]

async def index_posts(posts: list):
    """
    Generate embeddings and upsert to Qdrant.
    """
    async with httpx.AsyncClient() as client:
        for post in posts:
            embedding = await client.post(
                "http://embedding-service:8004/embed",
                json={"text": post["text"]})

            await client.put(
                "http://qdrant:6333/collections/posts/points",
                json={
                    "points": [{
                        "id": post["id"],
                        "vector": embedding.json()["embedding"],
                        "payload": post
                    }]
                }
            )
